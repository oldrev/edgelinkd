{{- /* Returns one entry per configured language for the language switchers: dict "name" "label" "locale" "href" "current" "href" points to the translation of the current page when it exists, otherwise to that language's home page. */ -}} {{- $page := . -}} {{- $links := dict -}} {{- range $page.AllTranslations -}} {{- $links = merge $links (dict .Language.Name .RelPermalink) -}} {{- end -}} {{- $out := slice -}} {{- range $page.AllTranslations -}} {{- $href := index $links .Language.Name | default (printf "%s/" .Language.Name | relURL) -}} {{- $out = $out | append (dict "name" .Language.Name "label" .Language.LanguageName "locale" .Language.Lang "href" $href "current" (eq .Language.Name site.Language.Name)) -}} {{- end -}} {{- return $out -}}
